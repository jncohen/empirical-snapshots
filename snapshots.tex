%%--- Vertical rhythm ------------------------------------------------
%% Calibrated against RENDERED ink, not source values: a \vspace is added to a
%% box that already carries the font's ascent/descent, so some are negative.
%% Target: ~6.2pt of air on each side of every rule.
\def\hflgapHeadRule{-1.8pt}   %% header text -> masthead rule
\def\hflgapRuleLogo{5pt}      %% masthead rule -> wordmark
\def\hflgapLogoWord{11pt}     %% HF square -> wordmark (horizontal)
%% Wordmark size drives the lockup. Bricolage's cap height measures 0.749 of
%% its point size, so the square is set to exactly that: the square's top and
%% bottom then land on the wordmark's cap line and baseline -- same band, no
%% overhang. Change \hflwmsize and the square follows.
\newlength{\hflwmsize}\setlength{\hflwmsize}{$if(snapshot_wordmark_size)$$snapshot_wordmark_size$$else$15pt$endif$}
\newlength{\hflboxsize}\setlength{\hflboxsize}{0.749\hflwmsize}
\newlength{\hflboxlift}\setlength{\hflboxlift}{0.268\hflboxsize}
\def\hflgapLogoTitle{4.9pt}  %% wordmark -> title  (no rule between)
\def\hflgapTitleRule{-4.7pt}  %% title -> closing rule
\def\hflgapRuleBody{11.75pt}    %% closing rule -> body
\def\hflgapFootRule{-10.05pt}  %% footer rule -> footer text
\def\hflavatarcol{0.26}       %% author portrait: share of the column width
\def\hflauthorcol{0.68}       %% author text block: share of the column width
\def\hflgapAvatarText{9pt}    %% portrait -> author text
%% The author box is pinned to the foot of column 2 by a \vfill, so it sits in
%% the bottom corner whatever the body length. These fix its two margins.
\def\hflgapAboveAuthor{0.12cm} %% rule -> "About the author"
\def\hflgapBelowAuthor{0.0cm} %% portrait bottom -> body bottom
